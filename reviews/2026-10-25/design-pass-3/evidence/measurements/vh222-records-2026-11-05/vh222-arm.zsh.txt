#!/bin/zsh
# VH-222 #5: arm one group by the turn rule (Orchestrator #40, 2026-11-05). At start, note each sibling (VH-220, VH-221, VH-223, VH-224, VH-231)
# with a go or wait script alive. Check $S/walk.lock/owner.txt every 60 s; a noted sibling's turn has passed once owner.txt names it, or once
# no script of its is alive. When every noted turn has passed, log the scripts and run go.sh <name> once (go.sh then waits for the lock, power
# and load as it stands). Past the deadline: one line, exit 10.
#   zsh vh222-arm.zsh <name> "<what for the lock>" <deadline, YYYY-MM-DDTHH:MM:SS UTC> '<job line>'
S=/Users/uazjob/UoN-Video-Helper-side; D=$S/vh222-walk; W=$S/vh222; LOCK=$S/walk.lock
WALKER="$W/reviews/2026-10-25/design-pass-3/evidence/measurements/walker"
NAME=$1; WHAT=$2; DL=$3; JOB=$4
LOG=$D/vh222-arm-$NAME.log
for f in "$LOG" "$D/$NAME" "$D/$NAME.go.out"; do [ -e "$f" ] && { echo "EXISTS: $f"; exit 2; }; done
DEADLINE=$(date -j -u -f '%Y-%m-%dT%H:%M:%S' "$DL" +%s) || exit 2
alive() { ps -axo command | grep -E '[g]o\.sh|[w]ait' | grep -v eval | grep -c "vh${1}" }
typeset -A WAIT
for n in 220 221 223 224 231; do [ "$(alive $n)" != 0 ] && WAIT[$n]=1; done
echo "$(date -u +%T) $NAME: waiting behind ${(k)WAIT:-none}" >> $LOG
LAST="-"
while :; do
  O=$(cat $LOCK/owner.txt 2>/dev/null)
  [ "$O" != "$LAST" ] && { echo "$(date -u +%T) owner: ${O:-none} | load $(uptime | sed -E 's/.*load averages?: //')" >> $LOG; LAST=$O; }
  for n in ${(k)WAIT}; do
    if print -r -- "$O" | grep -q "VH-$n "; then unset "WAIT[$n]"; echo "$(date -u +%T) VH-$n's turn seen" >> $LOG
    elif [ "$(alive $n)" = 0 ]; then unset "WAIT[$n]"; echo "$(date -u +%T) VH-$n has no script alive: passed" >> $LOG; fi
  done
  [ ${#WAIT} = 0 ] && break
  [ $(date +%s) -ge $DEADLINE ] && { echo "$(date -u +%T) DEADLINE, still behind ${(k)WAIT}" >> $LOG; echo "NOT ARMED by $DL: behind ${(k)WAIT}; owner: ${O:-none}"; exit 10; }
  sleep 60
done
echo "$(date -u +%T) armed; the go.sh and wait scripts now:" >> $LOG
ps -axo pid,command | grep -E '[g]o\.sh|[w]ait-seq' | grep -v 'eval' | cut -c1-200 >> $LOG
cd $D && zsh go.sh "$NAME" "$WHAT" "$WALKER" "$W" "$JOB" >> "$D/$NAME.go.out" 2>&1
rc=$?
echo "$(date -u +%T) go exit $rc" >> $LOG
echo "go exit $rc"; tail -3 "$D/$NAME.go.out"
