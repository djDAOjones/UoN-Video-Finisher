#!/bin/zsh
# VH-230 #1, copied from VH-218 #2: a side's groups, in sequence, one at a time; stops at the first non-zero exit. Usage: zsh seq.zsh before|after <walker> <app> [groups…]
side=$1; W=$2; A=$3; shift 3
D=/Users/joe/UoN-Video-Helper-side/vh230-walk; cd "$D"
groups=("$@"); [ ${#groups} = 0 ] && groups=(passed pending warned failed s103)
for g in $groups; do
  case "$g" in passed) s=S-82;; pending) s=S-82-pending;; warned) s=S-82-warned;; failed) s=S-82-failed;; s103) s=S-103;; *) echo "unknown $g"; exit 2;; esac
  zsh go.sh "$side-$g" "$side $g" "$W" "$A" "node vh230-driver.mjs walk --walker \"\$WALKER\" --label $side --name $g --scenarios $s" > "$side-$g.go.out" 2>&1; rc=$?
  echo "$side-$g exit $rc $(date -u +%T)" >> "$side.seq.log"
  [ $rc = 0 ] || { echo "SEQ STOP $side-$g rc $rc" >> "$side.seq.log"; exit $rc; }
done
echo "SEQ DONE $side" >> "$side.seq.log"
