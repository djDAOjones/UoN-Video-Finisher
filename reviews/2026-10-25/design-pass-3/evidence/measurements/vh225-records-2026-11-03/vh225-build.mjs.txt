const out = []
let bad = 0
for (const [file, swaps] of plan) {
  let text = readFileSync(file, 'utf8')
  for (const [from, to, n] of swaps) {
    const got = text.split(from).length - 1
    if (got !== n) { console.log(`COUNT ${file}: ${JSON.stringify(from)} ${got}, expected ${n}`); bad++ }
    text = text.split(from).join(to)
  }
  out.push([file, text])
}
if (bad) { console.log(`STOP: ${bad} counts differ; nothing written`); process.exit(1) }
if (process.argv[2] !== '--write') { console.log(`dry run OK: ${out.length} files, every count as expected`); process.exit(0) }
for (const [file, text] of out) writeFileSync(file, text)
console.log(`written ${out.length} files`)
