// VH-225 #1's driver, from VH-210's vh210-driver.mjs and vh210-driver-2.mjs (copied beside it, the originals untouched): the item's cells by the pass's
// own runner (matrix.mjs), unchanged, on a plan in memory, with a read-only name probe after each scenario's own `afterMode`.
// The probe reads, per cell: the tab title and its history (window.__audit.titles), the h1's text, box, lines, words broken inside, the title's "?" beside it,
// the statement's covers line, every mailto the page prepared (window.__mx.mailto) and every mailto link's subject, the page's sideways scroll, and, in ms-MY,
// three candidate words measured in the h1's own computed font against the title row's width (the design compare's settlement 2). The probe adds one hidden
// span for that measure and removes it; nothing else is written to the page.
// In memory only: `S-03:open` is S-03 with the title's "?" pressed at each load (driver-2's OPEN_PANEL); `--plan cross` crosses the scenarios with the
// languages, the views PLANS declares for each, and the modes; `--plan smoke` is PLANS.smoke as declared; `--plan titles` the job scenarios in one cell each.
//   APP_URL=… CDP_PORT=… FIX=… OUT_ROOT=… RUN_ROOT=… OUTD=… node vh225-driver.mjs walk --walker <dir> --label before --name <group> --plan cross --scenarios S-03,S-03:open --modes <list> [--langs] [--rm]
import { pathToFileURL, fileURLToPath } from 'node:url'
import { join, resolve } from 'node:path'
import { appendFileSync, mkdirSync } from 'node:fs'
import { execFileSync } from 'node:child_process'

const list = (s) => String(s ?? '').split(',').map((x) => x.trim()).filter(Boolean)
const args = (() => { const a = process.argv.slice(2), o = { _: [] }; for (let i = 0; i < a.length; i++) { if (a[i].startsWith('--')) { const k = a[i].slice(2); if (a[i + 1] === undefined || a[i + 1].startsWith('--')) o[k] = true; else o[k] = a[++i] } else o._.push(a[i]) } return o })()

export const MODES_MAIN = 'light,dark,forced-light,forced-dark,font32,dark-font32,forced-light-font32,forced-dark-font32,dark-spacing,forced-light-spacing,forced-dark-spacing'
export const MODES_SPFB = 'spacing,fallback'
const OPEN_PANEL = `addEventListener('load', () => { const b = document.getElementById('about-help-button'); if (b && b.getAttribute('aria-expanded') === 'false') b.click() })`
const WORDS = ['Pembantu', 'Penyiap', 'Penyempurna']

export const NAME_PROBE = `(() => {
  const round = (n) => Math.round(n * 100) / 100, de = document.documentElement, q = (s) => document.querySelector(s)
  const box = (el) => { if (!el) return null; const r = el.getBoundingClientRect(); return { l: round(r.left), t: round(r.top), r: round(r.right), b: round(r.bottom), w: round(r.width), h: round(r.height) } }
  const rects = (range) => [...range.getClientRects()].filter((x) => x.width > 0 && x.height > 0)
  const lines = (el) => { const r = document.createRange(); r.selectNodeContents(el); return new Set(rects(r).map((x) => Math.round(x.top))).size }
  const broken = (el) => { const out = [], tw = document.createTreeWalker(el, NodeFilter.SHOW_TEXT); let n
    while ((n = tw.nextNode())) { const re = /\\S+/g; let m; while ((m = re.exec(n.data))) { const r = document.createRange(); r.setStart(n, m.index); r.setEnd(n, m.index + m[0].length); if (new Set(rects(r).map((x) => Math.round(x.top))).size > 1) out.push({ word: m[0], cjk: /[\\u3400-\\u9fff]/.test(m[0]) }) } }
    return out }
  const text = (el) => (el ? el.textContent.replace(/\\s+/g, ' ').trim() : null)
  const shown = (el) => !!el && el.getClientRects().length > 0 && getComputedStyle(el).visibility !== 'hidden' && !el.closest('details:not([open])') && (!el.checkVisibility || el.checkVisibility({ visibilityProperty: true }))
  const h1 = q('h1'), btn = q('#about-help-button'), row = q('.page-title-row'), cov = q('[data-i18n="statement.covers"]')
  const out = { lang: de.lang, title: document.title, titles: [...new Set(((window.__audit || {}).titles || []).map((t) => t.title))],
    page: { cw: de.clientWidth, sw: de.scrollWidth, hscroll: de.scrollWidth > de.clientWidth, fontPx: parseFloat(getComputedStyle(de).fontSize) } }
  if (h1) { const cs = getComputedStyle(h1); out.h1 = { text: text(h1), shown: shown(h1), box: box(h1), lines: lines(h1), broken: broken(h1), overflow: h1.scrollWidth > h1.clientWidth + 0.5, font: cs.fontSize, lh: cs.lineHeight } }
  if (btn && h1) { const b = box(btn), hb = box(h1); out.help = { box: b, expanded: btn.getAttribute('aria-expanded'), beside: !!(b && hb && b.t < hb.b && b.b > hb.t), inside: !!b && b.l >= -0.5 && b.r <= de.clientWidth + 0.5 } }
  if (row) { const cs = getComputedStyle(row); out.row = { box: box(row), contentW: round(row.clientWidth - parseFloat(cs.paddingLeft) - parseFloat(cs.paddingRight)) } }
  if (cov) out.covers = { text: text(cov), shown: shown(cov), box: shown(cov) ? box(cov) : null, lines: shown(cov) ? lines(cov) : null, broken: shown(cov) ? broken(cov) : [] }
  out.mailto = (((window.__mx || {}).mailto) || []).map((m) => ({ to: m.to, subjectText: m.subjectText }))
  out.links = [...document.querySelectorAll('a[href^="mailto:"]')].map((a) => { let s = null; try { s = new URL(a.href).searchParams.get('subject') } catch {} return { sel: a.id ? '#' + a.id : (a.closest('[id]') || {}).id || 'a', subject: s } }).filter((l) => l.subject)
  if (h1 && de.lang === 'ms-MY') { const cs = getComputedStyle(h1), s = document.createElement('span'); s.setAttribute('aria-hidden', 'true')
    s.style.cssText = 'position:absolute;visibility:hidden;white-space:nowrap;left:0;top:0;font:' + cs.font + ';letter-spacing:' + cs.letterSpacing + ';word-spacing:' + cs.wordSpacing + ';text-transform:' + cs.textTransform
    h1.parentElement.append(s); out.words = {}; for (const w of ${JSON.stringify(WORDS)}) { s.textContent = w; out.words[w] = round(s.getBoundingClientRect().width) } s.remove() }
  return out })()`

function power() {
  const sh = (cmd, a) => execFileSync(cmd, a, { encoding: 'utf8' })
  const batt = sh('pmset', ['-g', 'batt']), pm = sh('pmset', ['-g']), up = sh('uptime', []).trim()
  const ac = /AC Power/.test(batt), level = Number((batt.match(/(\d+)%/) ?? [])[1]), mode = Number((pm.match(/^\s*powermode\s+(\d+)/m) ?? [])[1])
  return { at: new Date().toISOString(), source: ac ? 'AC' : 'battery', level, powermode: mode, load: Number((up.match(/load averages?: ([\d.]+)/) ?? [])[1]), ok: ac || (level > 50 && !(/lowpowermode\s+1/.test(pm) || mode === 1)) }
}

// The views PLANS declares for a scenario (every plan, parts registered), in the walker's order.
function declaredViews(PLANS, VIEWS, id) {
  const seen = new Set()
  for (const fn of Object.values(PLANS)) { let recs; try { recs = fn() } catch { continue } for (const r of recs ?? []) if (r.id === id) seen.add(r.view) }
  return Object.keys(VIEWS).filter((v) => seen.has(v))
}
function declaresRM(PLANS, id) {
  for (const fn of Object.values(PLANS)) { let recs; try { recs = fn() } catch { continue } if ((recs ?? []).some((r) => r.id === id && r.mode === 'reduced-motion')) return true }
  return false
}

async function build(walker) {
  const at = (f) => import(pathToFileURL(join(walker, f)).href)
  const S = await at('scenarios.mjs'); await at('parts.mjs')
  const { VIEWS } = await at('qualify.mjs')
  const plan = args.plan ?? 'cross'
  let records
  if (plan === 'cross') {
    const modes = list(args.modes ?? MODES_MAIN)
    records = list(args.scenarios).flatMap((spec) => {
      const [id, variant] = spec.split(':'), sc = S.SCENARIOS[id]
      if (!sc) throw new Error('no scenario ' + id)
      const langs = list(args.langs ?? 'en-GB,zh-Hans,ms-MY').filter((l) => !sc.langs || sc.langs.includes(l))
      const views = declaredViews(S.PLANS, VIEWS, id).filter((v) => !args.views || list(args.views).includes(v))
      const rm = args.rm && declaresRM(S.PLANS, id) ? langs.map((lang) => ({ id, lang, view: 'w1280', mode: 'reduced-motion' })) : []
      return [...langs.flatMap((lang) => views.flatMap((view) => modes.map((mode) => ({ id, lang, view, mode })))), ...rm].map((r) => (variant ? { ...r, variant } : r))
    })
  } else if (plan === 'smoke') records = S.PLANS.smoke().filter((r) => !list(args.skip).includes(r.id))
  else if (plan === 'titles') records = list(args.scenarios ?? 'S-48,S-53,S-56,S-52').flatMap((id) => list(args.langs ?? 'en-GB,zh-Hans,ms-MY').map((lang) => ({ id, lang, view: 'w1280', mode: 'light' })))
  else throw new Error('plan ' + plan)
  return { S, records }
}

async function walk() {
  for (const k of ['APP_URL', 'CDP_PORT', 'FIX', 'OUT_ROOT', 'RUN_ROOT', 'OUTD']) if (!process.env[k]) throw new Error('set ' + k)
  const walker = resolve(args.walker), label = args.label ?? 'before'
  const matrixUrl = pathToFileURL(join(walker, 'matrix.mjs')).href
  const at = (f) => import(new URL('./' + f, matrixUrl).href)
  const { S, records } = await build(walker)
  const { SCENARIOS, PLANS } = S
  const { sleep } = await at('walk.mjs')
  const plan = `vh225-${label}-${args.name ?? 'run'}`, p = power()
  mkdirSync(process.env.RUN_ROOT, { recursive: true })
  appendFileSync(join(process.env.RUN_ROOT, 'vh225-ledger.jsonl'), JSON.stringify({ plan, label, power: p, walker, records: records.length, args }) + '\n')
  if (!p.ok) { console.log('POWER-STOP'); process.exit(3) }
  const OUT = join(process.env.RUN_ROOT, `vh225-probe-${plan}.jsonl`)
  const wrap = (id, vid, orig) => async (w) => {
    // VH-219's in-memory spacing modes (as VH-210's driver): the prefix sets the colours, this adds the walker's own 1.4.12 sheet.
    if (/.-spacing$/.test(w.qualMode)) { await w.spacing(true); await sleep(100) }
    if (orig) await orig(w)
    let g
    try { g = await w.evaluate(NAME_PROBE) } catch (e) { g = { error: String(e.message ?? e).slice(0, 200) } }
    appendFileSync(OUT, JSON.stringify({ plan, cell: w.name.split('/').pop(), walk: w.name, id, ...(vid ? { variant: vid } : {}), mode: w.qualMode, at: new Date().toISOString(), ...g }) + '\n')
  }
  for (const id of [...new Set(records.map((r) => r.id))]) {
    const orig = SCENARIOS[id]; if (!orig) continue // a route: the walker's own records only, no hook
    SCENARIOS[id] = { ...orig, afterMode: wrap(id, null, orig.afterMode) }
    const vs = { ...(orig.variants ?? {}) }
    if (records.some((r) => r.id === id && r.variant === 'open')) vs.open = { name: `${orig.name}, the title's "?" open (VH-225's staged state)`, setup: async (w, lang) => { await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: OPEN_PANEL }); return orig.setup(w, lang) } }
    if (Object.keys(vs).length) SCENARIOS[id].variants = Object.fromEntries(Object.entries(vs).map(([v, x]) => [v, { ...x, afterMode: wrap(id, v, x.afterMode ?? orig.afterMode) }]))
  }
  PLANS[plan] = () => records.map((r) => ({ ...r }))
  process.argv = [process.argv[0], fileURLToPath(matrixUrl), plan]
  await import(matrixUrl)
}

// Read-only: the plan's records, by `node vh225-driver.mjs count --walker <dir> --plan … [--scenarios …] [--modes …] [--rm]`.
async function count() {
  const { records } = await build(resolve(args.walker))
  const by = {}; for (const r of records) { const k = r.id + (r.variant ? ':' + r.variant : ''); by[k] = (by[k] ?? 0) + 1 }
  console.log('records', records.length, JSON.stringify(by))
}
if (args._[0] === 'walk') await walk()
else if (args._[0] === 'count') await count()
else if (args._[0] !== 'lib') { console.log('usage: node vh225-driver.mjs walk|count --walker <dir> [--label l --name n] --plan cross|smoke|titles [--scenarios …] [--modes …] [--langs …] [--rm] [--skip …]'); process.exit(2) }
