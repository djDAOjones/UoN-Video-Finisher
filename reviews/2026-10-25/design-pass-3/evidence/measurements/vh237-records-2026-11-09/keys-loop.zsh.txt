#!/bin/zsh
# VH-237 #1: a copy of VH-234's file, adapted only in the folder (vh237-walk), W, BASE, WHO ("VH-237 #1") and names.
# VH-237 #1: VH-203 #7's split-7.zsh loop mode (85cfd643), the original untouched: one driver run per key, so each
# key is its own setup; matrix.mjs and the walker unchanged. Changed: the driver's name (vh237-driver.mjs), no --plan (its default, every
# record of parts.mjs's RUNS, as VH-203's `--plan vh203-after`), and the chain mode left out (group.zsh queues).
#   zsh keys-loop.zsh <keys file> <label> <folder name>   (the job line; walk.sh runs it in vh237-walk with WALKER, OUTD … set)
kf=$1; label=$2; n=$3
for k in WALKER OUTD RUN_ROOT; do [ -n "${(P)k}" ] || { echo "set $k"; exit 10; }; done
K="$OUTD/keys"; mkdir -p "$K" "$OUTD/logs"; keys=(${(f)"$(<$kf)"}); i=0; worst=0
for key in $keys; do
  [ -n "$key" ] || continue
  i=$((i+1)); t=$(printf %03d $i); f="$K/$t-$key.txt"; print -r -- "$key" > "$f"
  echo "key $t $key start $(date -u +%FT%TZ) load $(uptime | sed -E 's/.*load averages?: ([0-9.]+).*/\1/')" >> "$OUTD/split.log"
  node vh237-driver.mjs walk --walker "$WALKER" --label "$label" --name "$n" --keys "$f" < /dev/null > "$OUTD/logs/key$t.log" 2>&1; rc=$?
  echo "key $t $key exit $rc $(date -u +%FT%TZ)" >> "$OUTD/split.log"
  [ $rc -gt $worst ] && worst=$rc
done
echo "keys $i worst exit $worst $(date -u +%FT%TZ)" >> "$OUTD/split.log"
exit $worst
