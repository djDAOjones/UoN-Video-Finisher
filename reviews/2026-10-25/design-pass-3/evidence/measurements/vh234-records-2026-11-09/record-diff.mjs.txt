// VH-234 #2: the 102 cells before and after, by key: each instrument whose whole record differs (outcome or not), counted, with the cells; read-only.
import { readdirSync, readFileSync } from 'node:fs'
import { join } from 'node:path'
const load = (dir) => { const m = new Map(); for (const r of readdirSync(dir)) { let fs; try { fs = readdirSync(join(dir, r)) } catch { continue }
  for (const f of fs.filter((f) => /^S-14\d__.*\.json$/.test(f))) { const j = JSON.parse(readFileSync(join(dir, r, f), 'utf8')); m.set(j.key, j) } } return m }
const [b, a] = [load(process.argv[2]), load(process.argv[3])], rec = {}, out = {}
for (const [k, jb] of b) { const ja = a.get(k); for (const i of new Set([...Object.keys(jb.instruments), ...Object.keys(ja.instruments)])) {
  if (JSON.stringify(jb.instruments[i]) !== JSON.stringify(ja.instruments[i])) { (rec[i] ??= []).push(k); if (jb.instruments[i]?.outcome !== ja.instruments[i]?.outcome) out[i] = (out[i] || 0) + 1 } } }
for (const [i, ks] of Object.entries(rec)) console.log(`${i}: record differs in ${ks.length}, outcome in ${out[i] || 0}: ${ks.join(' ')}`)
