// VH-218 #2: each row's longest unbroken word, before and after, in every S-82 cell. The before's probe kept the widest word and the broken ones; `before-words` (driver 3, the base by its export, one field added: every word's width) gives every candidate's width by language, width and text condition, and a cell's broken list removes the broken. Colour modes agree, so each state's cells give one line per colour mode in the jsonl.
import { readdirSync, readFileSync, writeFileSync } from 'node:fs'; import { join } from 'node:path'
const D = process.argv[2]; const rd = (g) => { const r = join(D, g, 'runs'); return readFileSync(join(r, readdirSync(r).find((x) => /probe/.test(x))), 'utf8').trim().split('\n').map(JSON.parse) }
const cond = (m) => (m.endsWith('font32') ? 'font32' : m.endsWith('spacing') ? 'spacing' : 'default'); const LS = ['en-GB', 'zh-Hans', 'ms-MY']; const lang = (c) => LS.find((l) => c.includes(l)); const view = (c) => 'w' + c.split('-w').at(-1)
const W = {}; let proofSame = 0, proofN = 0; const base = Object.fromEntries(rd('before-passed').map((j) => [j.cell + '/' + j.mode, j]))
for (const j of rd('before-words')) { for (const r of j.rows) W[[lang(j.cell), view(j.cell), cond(j.mode), r.id]] = Object.fromEntries(r.label.words.map((w) => [w.t, w.w]))
  const b = base[j.cell + '/' + j.mode]; proofN++; const noW = (x) => JSON.stringify(x, (k, v) => (k === 'words' ? undefined : v)); if (noW(j.rows) === noW(b.rows)) proofSame++ }
const lines = [], tab = {}
const longest = (j, r, side) => { const K = [lang(j.cell), view(j.cell), cond(j.mode), r.id]; const w = W[K]; if (!w) throw new Error('no words ' + K); const broken = new Set(r.label.broken.map((b) => b.t)); const u = Object.entries(w).filter(([t]) => !broken.has(t)).sort((a, b) => b[1] - a[1])[0]; return u ? { word: u[0], w: u[1] } : null }
for (const st of ['passed', 'pending', 'warned', 'failed']) for (const [side, g] of [['before', `before-${st}`], ['after', `after2-${st}`]]) for (const j of rd(g)) for (const r of j.rows) { const u = longest(j, r, side); lines.push(JSON.stringify({ side, state: st, lang: lang(j.cell), view: view(j.cell), mode: j.mode, row: r.id, lines: r.label.lines, longestUnbroken: u, broken: r.label.broken.map((b) => b.t) })); if (st === 'passed' && j.mode.match(/^(light|font32|spacing)$/)) ((tab[[lang(j.cell), view(j.cell), cond(j.mode)].join(' ')] ??= {})[r.id] ??= {})[side] = u }
writeFileSync(join(D, 'longest-unbroken.jsonl'), lines.join('\n') + '\n')
const nn = lines.map(JSON.parse), noneB = nn.filter((x) => x.side === 'before' && !x.longestUnbroken).length, noneA = nn.filter((x) => x.side === 'after' && !x.longestUnbroken).length
const ids = ['secure', 'webcodecs', 'h264', 'aac', 'opfs', 'worker']; const f = (u) => (u ? `${u.word} ${Math.round(u.w)}` : 'none')
const out = [`Each row's longest unbroken word (px), before → after, passed, light; every state and colour mode, ${nn.length} readings, in \`longest-unbroken.jsonl\`. "None" means every word in the label broke: ${noneB} readings before, ${noneA} after. The before's word widths are from \`before-words\`, 27 cells on the base by the export, its rows ${proofSame} of ${proofN} identical to \`before-passed\`.`, '', `| Language, width, text | ${ids.join(' | ')} |`, `| --- | ${ids.map(() => '---').join(' | ')} |`]
for (const k of Object.keys(tab)) out.push(`| ${k} | ${ids.map((i) => { const a = f(tab[k][i].before), b = f(tab[k][i].after); return a === b ? a : `${a} → ${b}` }).join(' | ')} |`)
console.log(out.join('\n'))
