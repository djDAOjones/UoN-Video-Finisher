// VH-220 #1's driver, from VH-225's vh225-driver.mjs (copied beside it, the original untouched): VH-150's six states by the pass's own runner
// (matrix.mjs), unchanged, on a plan in memory, with a read-only states probe after each scenario's own `afterMode`.
// The states: S-90b (a watchdog failure, F11), S-115 (a second job), S-30 (an unreadable file beside an unsaved result), S-138 (a kept result
// with warnings, F16 + F4d), S-31:recheck (a storage re-check that still fails, F3) and S-32:set-start / S-32:set-end (both Set buttons, F7).
// In memory only, as VH-225's S-03:open was: S-31:recheck is S-31's own setup, then Check again pressed and the check let finish into the
// block again; S-32:set-start and :set-end are `ready`, focus on that Set button, then F7 in place. Each variant's setup installs a
// MutationObserver on #source-status and #status before it acts (it writes nothing to the page); the probe reads it.
// `--uncited` drops every condition VH-199's records read (PLANS run-01 to run-12 less matrix-coverage-2026-10-27.json's missing keys).
// VH-220 #4's copy (vh220-driver-2.mjs; vh220-driver.mjs ae571d5a untouched), on Orchestrator #41's word, by S-31:recheck's pattern, adds two
// in-memory variants, matrix.mjs and every walker file unchanged:
//   S-115:afterCreate (R2-2's reset instant): S-115's own setup, run unchanged through a proxy of the walk object that, on the first Create
//     click after the save, installs a reader just before the click; the reader takes the bar, its label, the stage line, the time left and
//     the status line at the click, at the end of the click's own task (after the page's handler), at the next two frames and at every DOM
//     change after, each with its time from the click, until S-115's setup has seen encoding begin; then one end reading.
//   S-90b:langAfter (R2-1's language change): S-90b's own setup in en-GB, settled (locks lifted), then the page language changed to the
//     record's language by the switcher, then the status line, the job notice and the failure's words read, with every status text between.
// A scenario spec may carry its own language, `S-90b:langAfter@zh-Hans`, in place of --langs for that spec alone.
//   APP_URL=… CDP_PORT=… FIX=… OUT_ROOT=… RUN_ROOT=… OUTD=… node vh220-driver-2.mjs walk --walker <dir> --label before --name <group> --scenarios S-31:recheck,… [--modes …] [--langs …] [--views …] [--uncited]
//   node vh220-driver-2.mjs count --walker <dir> --scenarios … [--uncited]

function power() {
  const sh = (cmd, a) => execFileSync(cmd, a, { encoding: 'utf8' })
  const batt = sh('pmset', ['-g', 'batt']), pm = sh('pmset', ['-g']), up = sh('uptime', []).trim()
  const ac = /AC Power/.test(batt), level = Number((batt.match(/(\d+)%/) ?? [])[1]), mode = Number((pm.match(/^\s*powermode\s+(\d+)/m) ?? [])[1])
  return { at: new Date().toISOString(), source: ac ? 'AC' : 'battery', level, powermode: mode, load: Number((up.match(/load averages?: ([\d.]+)/) ?? [])[1]), ok: ac || (level > 50 && !(/lowpowermode\s+1/.test(pm) || mode === 1)) }
}

// VH-199's read conditions: PLANS run-01 to run-12, less every missing key in the coverage account.
function citedSet(walker, S) {
  const cov = JSON.parse(readFileSync(join(walker, '..', 'matrix-coverage-2026-10-27.json'), 'utf8'))
  const missing = new Set(cov.runs.flatMap((r) => r.missing.map((m) => m.key)))
  const read = new Set()
  for (let i = 1; i <= 12; i++) for (const r of S.PLANS['run-' + String(i).padStart(2, '0')]()) { const id = r.id + (r.variant ? ':' + r.variant : ''); const k = [id, r.lang, r.view, r.mode].join('__'); if (!missing.has(k)) read.add(k) }
  return read
}

async function build(walker) {
  const at = (f) => import(pathToFileURL(join(walker, f)).href)
  const S = await at('scenarios.mjs'); await at('parts.mjs')
  const modes = list(args.modes ?? MODES), views = list(args.views ?? VIEWS_IN), langs = list(args.langs ?? 'en-GB,zh-Hans,ms-MY')
  const cited = citedSet(walker, S)
  const all = list(args.scenarios ?? STATES).flatMap((spec) => { const [base, own] = spec.split('@'); const [id, variant] = base.split(':'); if (!S.SCENARIOS[id]) throw new Error('no scenario ' + id)
    return (own ? [own] : langs).flatMap((lang) => views.flatMap((view) => modes.map((mode) => ({ id, lang, view, mode, ...(variant ? { variant } : {}) })))) })
  const keyOf = (r) => [r.id + (r.variant ? ':' + r.variant : ''), r.lang, r.view, r.mode].join('__')
  const records = args.uncited ? all.filter((r) => !cited.has(keyOf(r))) : all
  return { S, records, all, cited: all.filter((r) => cited.has(keyOf(r))).map(keyOf) }
}

async function walk() {

  const walker = resolve(args.walker), label = args.label ?? 'before'

  const { S, records, cited } = await build(walker)
  const { SCENARIOS, PLANS, ready } = S
  const { sleep } = await at('walk.mjs')
  const { applyFault } = await at('faults.mjs')
  const plan = `vh220-${label}-${args.name ?? 'run'}`, p = power()
  mkdirSync(process.env.RUN_ROOT, { recursive: true })
  appendFileSync(join(process.env.RUN_ROOT, 'vh220-ledger.jsonl'), JSON.stringify({ plan, label, power: p, walker, records: records.length, cited, args }) + '\n')
  if (!p.ok) { console.log('POWER-STOP'); process.exit(3) }
  const OUT = join(process.env.RUN_ROOT, `vh220-probe-${plan}.jsonl`)
  const wrap = (id, vid, orig) => async (w) => {
    if (orig) await orig(w)
    let g
    try { g = await w.evaluate(STATES_PROBE) } catch (e) { g = { error: String(e.message ?? e).slice(0, 200) } }
    appendFileSync(OUT, JSON.stringify({ plan, cell: w.name.split('/').pop(), walk: w.name, id, ...(vid ? { variant: vid } : {}), mode: w.qualMode, at: new Date().toISOString(), ...g }) + '\n')
  }
  const MINE = {
    'S-31': { recheck: (orig) => ({ name: `${orig.name}, then Check again pressed and the re-check let finish: still blocked (VH-220's in-memory variant)`,
      state: `(${orig.state}) && !!window.__vh220 && window.__vh220.checking && !!window.__vh220.after`,
      setup: async (w, lang) => { await orig.setup(w, lang); await w.evaluate(WATCH); await w.click(CHECK_AGAIN)
        await w.waitFor(`window.__vh220.checking`, { label: 'the re-check running', timeout: 30000 })
        await w.waitFor(`(() => { const M = window.__mx; return !M.checking() && M.kindIn('#source-block') === 'error' && M.checkAgain() })()`, { label: 'blocked again', timeout: 120000 }); await sleep(400); await w.evaluate(AFTER) } }) },
    'S-32': Object.fromEntries(['start', 'end'].map((e) => ['set-' + e, (orig) => ({ name: `${orig.name}, with focus on "Set ${e} here" when the preview fails (VH-220's in-memory variant)`,
      state: `(${orig.state}) && !!window.__vh220 && !!window.__vh220.after && window.__vh220.before === 'trim-set-${e}'`,
      setup: async (w, lang) => { await ready(w, lang); await w.evaluate(WATCH); await w.evaluate(`(() => { document.querySelector('#trim-set-${e}').focus(); window.__vh220.before = document.activeElement.id; return window.__vh220.before })()`); await applyFault(w, 'F7'); await sleep(400); await w.evaluate(AFTER) } })])),
    // VH-220 #4: S-115's own setup, unchanged, through a proxy: after waitSaved, the next click on Create gets the reader first and a read after.
    'S-115': { afterCreate: (orig) => ({ name: `${orig.name}, read from the second job's Create click until encoding begins: the bar, its words and the time left (VH-220's in-memory variant)`,
      state: `(${orig.state}) && !!window.__vh220 && window.__vh220.kind === 'afterCreate' && !!window.__vh220.click && !!window.__vh220.checks`,

// Read-only: the plan's records and the cited conditions.
async function count() {
  const { records, all, cited } = await build(resolve(args.walker))
  const by = {}; for (const r of records) { const k = r.id + (r.variant ? ':' + r.variant : ''); by[k] = (by[k] ?? 0) + 1 }
  console.log('conditions', all.length, 'records', records.length, JSON.stringify(by), 'cited', cited.length, cited.join(' '))
}
if (args._[0] === 'walk') await walk()
else if (args._[0] === 'count') await count()
else if (args._[0] !== 'lib') { console.log('usage: node vh220-driver-2.mjs walk|count --walker <dir> [--label l --name n] [--scenarios …] [--modes …] [--langs …] [--views …] [--uncited]'); process.exit(2) }
