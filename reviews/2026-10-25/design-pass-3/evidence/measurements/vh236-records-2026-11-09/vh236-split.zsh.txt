#!/bin/zsh
# VH-236 #1: the loop of vh203-walk/split-7.zsh (its chain mode not taken), one driver run per key, so that each S-149 key is its own setup
# (S-149 is focus-bound: focus-split-10.mjs's FOCUS); the driver vh236-driver.mjs; matrix.mjs and the walker unchanged.
# Usage (the job line; walk.sh runs it in vh236-walk with WALKER, OUTD, RUN_ROOT … set): zsh vh236-split.zsh loop <keys file> <label> <folder name>
mode=$1; shift
if [ "$mode" = loop ]; then
  kf=$1; label=$2; n=$3
  for k in WALKER OUTD RUN_ROOT; do [ -n "${(P)k}" ] || { echo "set $k"; exit 10; }; done
  K="$OUTD/keys"; mkdir -p "$K" "$OUTD/logs"; keys=(${(f)"$(<$kf)"}); i=0; worst=0
  for key in $keys; do
    [ -n "$key" ] || continue
    i=$((i+1)); t=$(printf %02d $i); f="$K/$t-$key.txt"; print -r -- "$key" > "$f"
    echo "key $t $key start $(date -u +%FT%TZ) load $(uptime | sed -E 's/.*load averages?: ([0-9.]+).*/\1/')" >> "$OUTD/split.log"
    node vh236-driver.mjs walk --walker "$WALKER" --label "$label" --name "$n" --keys "$f" < /dev/null > "$OUTD/logs/key$t.log" 2>&1; rc=$?
    echo "key $t $key exit $rc $(date -u +%FT%TZ)" >> "$OUTD/split.log"
    [ $rc -gt $worst ] && worst=$rc
  done
  echo "keys $i worst exit $worst $(date -u +%FT%TZ)" >> "$OUTD/split.log"
  exit $worst
fi
echo "usage: zsh vh236-split.zsh loop <keys file> <label> <folder name>"; exit 2
