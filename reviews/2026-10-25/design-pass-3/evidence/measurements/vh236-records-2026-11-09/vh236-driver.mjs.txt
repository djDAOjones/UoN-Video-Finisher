// VH-236 #1's driver: VH-203's vh203-driver.mjs (2a9671cd, copied here as vh203-driver.mjs.orig, the original untouched), cut to one plan.
// It registers PLANS['vh236'] in memory, from existing scenario definitions only (no new scenario or ID; the walker unchanged), the item's
// 52 cells: S-28 x en-GB, zh-Hans, ms-MY x w1280, w375, w320, light (9); S-140n at w1280, en-GB, light (1); S-29 as S-28 (9); the control
// S-32 as S-28 (9); S-149 x the three languages x w1280, w320 x light, dark, forced-light, forced-dark (24, VH-207's and VH-203 b3's
// conditions). `matrix.mjs vh236 --keys <file>` (VH-223's interface) then runs as declared. It records power, both loads and the key list
// in a ledger line, and refuses off power.
//   APP_URL=… CDP_PORT=… FIX=… OUT_ROOT=… RUN_ROOT=… OUTD=… node vh236-driver.mjs walk --walker <dir> --label <l> --name <group> --keys <file>
//   RUN_ROOT=<any> node vh236-driver.mjs keys --walker <dir> --out <file>     (read-only: every key of the plan, one a line)
import { pathToFileURL, fileURLToPath } from 'node:url'
import { join, resolve } from 'node:path'
import { appendFileSync, existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs'
import { execFileSync } from 'node:child_process'
import { createHash } from 'node:crypto'

const args = (() => { const a = process.argv.slice(2), o = { _: [] }; for (let i = 0; i < a.length; i++) { if (a[i].startsWith('--')) { const k = a[i].slice(2); if (a[i + 1] === undefined || a[i + 1].startsWith('--')) o[k] = true; else o[k] = a[++i] } else o._.push(a[i]) } return o })()
const PLAN = 'vh236'
const L = ['en-GB', 'zh-Hans', 'ms-MY'], V3 = ['w1280', 'w375', 'w320'], MO = ['light', 'dark', 'forced-light', 'forced-dark']
const light3 = (id) => L.flatMap((lang) => V3.map((view) => ({ id, lang, view, mode: 'light' })))
const CELLS = [
  ...light3('S-28'),
  { id: 'S-140n', lang: 'en-GB', view: 'w1280', mode: 'light' },
  ...light3('S-29'),
  ...light3('S-32'),
  ...L.flatMap((lang) => ['w1280', 'w320'].flatMap((view) => MO.map((mode) => ({ id: 'S-149', lang, view, mode })))),
]

function power() {
  const sh = (cmd, a) => execFileSync(cmd, a, { encoding: 'utf8' })
  const batt = sh('pmset', ['-g', 'batt']), pm = sh('pmset', ['-g']), up = sh('uptime', []).trim()
  const ac = /AC Power/.test(batt), level = Number((batt.match(/(\d+)%/) ?? [])[1]), mode = Number((pm.match(/^\s*powermode\s+(\d+)/m) ?? [])[1])
  const la = up.match(/load averages?: ([\d.]+),? ([\d.]+),? ([\d.]+)/) ?? []
  return { at: new Date().toISOString(), source: ac ? 'AC' : 'battery', level, powermode: mode, load: Number(la[1]), load5: Number(la[2]), load15: Number(la[3]), ok: ac || (level > 50 && !(/lowpowermode\s+1/.test(pm) || mode === 1)) }
}
const keyFile = () => { if (typeof args.keys !== 'string') throw new Error('--keys <file> is required'); return resolve(args.keys) }

async function register(walker) {
  const at = (f) => import(pathToFileURL(join(walker, f)).href)
  const S = await at('scenarios.mjs'), { recordKey } = await at('runner.mjs')
  for (const r of CELLS) if (!S.SCENARIOS[r.id]) throw new Error('the walker holds no ' + r.id)
  S.PLANS[PLAN] = () => CELLS.map((r) => ({ ...r }))
  return { records: S.PLANS[PLAN](), recordKey }
}

async function walk() {
  for (const k of ['APP_URL', 'CDP_PORT', 'FIX', 'OUT_ROOT', 'RUN_ROOT', 'OUTD']) if (!process.env[k]) throw new Error('set ' + k)
  const walker = resolve(args.walker), label = args.label ?? 'after', file = keyFile(), bytes = readFileSync(file)
  const keys = String(bytes).split('\n').map((l) => l.trim()).filter(Boolean)
  await register(walker)
  const matrixUrl = pathToFileURL(join(walker, 'matrix.mjs')).href, p = power()
  mkdirSync(process.env.RUN_ROOT, { recursive: true })
  appendFileSync(join(process.env.RUN_ROOT, 'vh236-ledger.jsonl'), JSON.stringify({ label, name: args.name, plan: PLAN, power: p, walker, keys: { file, count: keys.length, sha1: createHash('sha1').update(bytes).digest('hex') }, args }) + '\n')
  if (!p.ok) { console.log('POWER-STOP'); process.exit(3) }
  process.argv = [process.argv[0], fileURLToPath(matrixUrl), PLAN, '--keys', file]
  await import(matrixUrl)
}

async function keysOut() {
  const walker = resolve(args.walker), { records, recordKey } = await register(walker)
  if (typeof args.out !== 'string') throw new Error('--out <file> is required')
  if (existsSync(resolve(args.out))) throw new Error('EXISTS ' + args.out)
  writeFileSync(resolve(args.out), records.map((r) => recordKey(r)).join('\n') + '\n')
  console.log(PLAN, records.length, 'keys to', args.out)
}
if (args._[0] === 'walk') await walk()
else if (args._[0] === 'keys') await keysOut()
else { console.log('usage: node vh236-driver.mjs walk|keys --walker <dir> [--label l --name n --keys <file>] | --out <file>'); process.exit(2) }
